// Guided customer journey: automatic movement only while the customer is inactive. // Customer activity RESTARTS the active inactivity clock; it does not permanently cancel the journey. let guidedScrollInProgress = false; let openingGuideTimer = null; let photoGuideTimer = null; let ideaGuideTimer = null; let openingGuideCancelled = false; let photoGuideActive = false; function smoothScrollToTarget(targetId, block = 'start', after) { const target = document.getElementById(targetId); if (!target) return; guidedScrollInProgress = true; target.scrollIntoView({ behavior: 'smooth', block }); setTimeout(() => { guidedScrollInProgress = false; if (typeof after === 'function') { after(); } }, 750); } function cancelOpeningGuide() { openingGuideCancelled = true; if (openingGuideTimer) { clearTimeout(openingGuideTimer); } } function schedulePhotoGuideAdvance() { if (photoGuideTimer) { clearTimeout(photoGuideTimer); } if (!photoGuideActive) return; photoGuideTimer = setTimeout(() => { if (!photoGuideActive) return; photoGuideActive = false; smoothScrollToTarget('ideaCard', 'center', () => { const idea = document.getElementById('ideaDesc'); if (idea) { idea.focus({ preventScroll: true }); } restartIdeaGuidanceTimer(); }); }, 10000); } function noteRealCustomerActivity() { if (guidedScrollInProgress) return; cancelOpeningGuide(); // Every customer interaction restarts a completely new 10-second clock. // It does not permanently cancel the movement to the description section. if (photoGuideActive) { schedulePhotoGuideAdvance(); } } ['wheel', 'touchstart', 'pointerdown', 'keydown'].forEach(eventName => { window.addEventListener(eventName, noteRealCustomerActivity, { passive: true }); }); function startOpeningGuide() { openingGuideCancelled = false; if (openingGuideTimer) { clearTimeout(openingGuideTimer); } openingGuideTimer = setTimeout(() => { if (!openingGuideCancelled) { smoothScrollToTarget('uploadPhotoCard', 'start'); } }, 12000); } function startPhotoGuide() { photoGuideActive = true; if (photoGuideTimer) { clearTimeout(photoGuideTimer); } // First position the uploaded photo so the Start Over control is visible. setTimeout(() => { if (photoGuideActive) { smoothScrollToTarget('startOverBtn', 'end'); } }, 150); // Begin the first 10-second inactivity countdown. schedulePhotoGuideAdvance(); }
